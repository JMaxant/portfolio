{{- /* /llms.txt, per https://llmstxt.org: title, summary, then one list of links per section.
       Generated from the content, so it never needs rewriting. See docs/seo.md#llmstxt. */ -}}
{{- define "partials/llms-link.txt" -}}
{{- /* The Markdown twin when the page has one: a cleaner target than the HTML. */ -}}
{{- $url := .Permalink -}}
{{- with .OutputFormats.Get "markdown" }}{{ $url = .Permalink }}{{ end -}}
- [{{ .Title }}]({{ $url }}){{ with or .Description .Summary }}: {{ . | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace }}{{ end }}
{{- end -}}

{{- $sections := slice -}}
{{- range site.Params.mainSections -}}
  {{- with site.GetPage (printf "/%s" .) -}}
    {{- with .RegularPages -}}
      {{- $sections = $sections | append (dict "title" (index . 0).CurrentSection.Title "pages" .) -}}
    {{- end -}}
  {{- end -}}
{{- end -}}
{{- /* The resume and its section, then the veille as a whole: its entries link elsewhere. */ -}}
{{- with site.GetPage "/a-propos" -}}
  {{- $sections = $sections | append (dict "title" .Title "pages" ((slice .) | append .RegularPages)) -}}
{{- end -}}
{{- with site.GetPage "/veille" -}}
  {{- $sections = $sections | append (dict "title" "Optional" "pages" (slice .)) -}}
{{- end -}}

> {{ site.Params.description }}

{{ . | strings.TrimSpace }}
{{- end }}
